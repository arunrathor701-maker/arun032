/* ==================================================
   NEXOFFERS API
================================================== */

async function loadOffers(){

    const table =
        document
        .getElementById(
            "offersTable"
        );


    table.innerHTML = `

        <tr>

            <td
                colspan="5"
                class="empty">

                Loading offers...

            </td>

        </tr>

    `;


    /*
    Do not request if token
    has not been entered.
    */

    if(
        API_TOKEN ===
        "YOUR_API_TOKEN"
    ){

        table.innerHTML = `

            <tr>

                <td
                    colspan="5"
                    class="empty">

                    Add your NexOffers API token
                    in the JavaScript settings.

                </td>

            </tr>

        `;

        return;

    }


    try{

        const url =
            API_BASE +
            encodeURIComponent(
                API_TOKEN
            ) +
            "/offers";


        const response =
            await fetch(
                url,
                {
                    method:"GET"
                }
            );


        if(!response.ok){

            throw new Error(
                "API Error: " +
                response.status
            );

        }


        const result =
            await response.json();


        /*
        Try common response formats.
        */

        if(
            Array.isArray(result)
        ){

            offers = result;

        }

        else if(
            Array.isArray(
                result.offers
            )
        ){

            offers =
                result.offers;

        }

        else if(
            Array.isArray(
                result.data
            )
        ){

            offers =
                result.data;

        }

        else{

            offers = [];

        }


        renderOffers();

        updateDashboard();

    }

    catch(error){

        console.error(error);


        table.innerHTML = `

            <tr>

                <td
                    colspan="5"
                    class="empty">

                    NexOffers API connection failed.

                    <br><br>

                    ${escapeHTML(
                        error.message
                    )}

                </td>

            </tr>

        `;

    }

}


/* ==================================================
   RENDER OFFERS
================================================== */

function renderOffers(){

    const table =
        document
        .getElementById(
            "offersTable"
        );


    if(
        !offers.length
    ){

        table.innerHTML = `

            <tr>

                <td
                    colspan="5"
                    class="empty">

                    No offers found.

                </td>

            </tr>

        `;

        return;

    }


    table.innerHTML =
        offers
        .map(
            offer => {

                const id =
                    offer.id ??
                    offer.offer_id ??
                    "-";


                const name =
                    offer.name ??
                    offer.title ??
                    offer.offer_name ??
                    "Offer";


                const category =
                    offer.category ??
                    "-";


                const payout =
                    offer.payout ??
                    offer.commission ??
                    offer.amount ??
                    0;


                const status =
                    offer.status ??
                    "Active";


                return `

                    <tr>

                        <td>
                            ${escapeHTML(id)}
                        </td>

                        <td>
                            ${escapeHTML(name)}
                        </td>

                        <td>
                            ${escapeHTML(category)}
                        </td>

                        <td>
                            ₹${Number(payout)
                            .toLocaleString("en-IN")}
                        </td>

                        <td>

                            <span
                                class="status active-status">

                                ${escapeHTML(
                                    status
                                )}

                            </span>

                        </td>

                    </tr>

                `;

            }
        )
        .join("");

}


/* ==================================================
   DASHBOARD
================================================== */

function updateDashboard(){

    let clicks = 0;

    let conversions = 0;

    let earnings = 0;


    members.forEach(
        member => {

            clicks +=
                Number(
                    member.clicks || 0
                );


            conversions +=
                Number(
                    member.conversions || 0
                );


            earnings +=
                Number(
                    member.earnings ||
                    member.commission ||
                    0
                );

        }
    );


    document
    .getElementById(
        "memberCount"
    )
    .textContent =
        members.length;


    document
    .getElementById(
        "clickCount"
    )
    .textContent =
        clicks.toLocaleString(
            "en-IN"
        );


    document
    .getElementById(
        "conversionCount"
    )
    .textContent =
        conversions.toLocaleString(
            "en-IN"
        );


    document
    .getElementById(
        "earningCount"
    )
    .textContent =
        "₹" +
        earnings.toLocaleString(
            "en-IN"
        );


    document
    .getElementById(
        "offerCount"
    )
    .textContent =
        offers.length;


    document

                    : "none";

            }
        );

    }
);

function escapeHTML(value){

    return String(
        value ?? ""
    )
    .replace(
        /&/g,
        "&amp;"
    )
    .replace(
        /</g,
        "&lt;"
    )
    .replace(
        />/g,
        "&gt;"
    )
    .replace(
        /"/g,
        "&quot;"
    )
    .replace(
        /'/g,
        "&#039;"
    );

}
